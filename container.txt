name: Import Container TAR from Storage to ACR

on:
  workflow_dispatch:
    inputs:
      blob_name:
        description: "Blob path and TAR filename"
        required: true
        type: string
        default: "myapp/myapp-1.0.0.tar"

      acr_repository:
        description: "Destination ACR repository"
        required: true
        type: string
        default: "apps/myapp"

      image_tag:
        description: "Destination ACR image tag"
        required: true
        type: string
        default: "1.0.0"

permissions:
  contents: read

env:
  # GitHub repository or organization variables
  STORAGE_ACCOUNT_NAME: ${{ vars.STORAGE_ACCOUNT_NAME }}
  STORAGE_CONTAINER_NAME: ${{ vars.STORAGE_CONTAINER_NAME }}
  ACR_NAME: ${{ vars.ACR_NAME }}

  # Offline-installed executables
  DOCKER_EXE: 'C:\Program Files\Docker\docker.exe'
  AZ_EXE: 'C:\Program Files\Microsoft SDKs\Azure\CLI2\wbin\az.cmd'

jobs:
  import-image:
    name: Import TAR to ACR

    runs-on:
      - self-hosted
      - windows
      - container-import

    timeout-minutes: 90

    steps:
      - name: Validate runner prerequisites
        shell: powershell
        run: |
          $ErrorActionPreference = "Stop"

          $requiredVariables = @{
              STORAGE_ACCOUNT_NAME   = $env:STORAGE_ACCOUNT_NAME
              STORAGE_CONTAINER_NAME = $env:STORAGE_CONTAINER_NAME
              ACR_NAME               = $env:ACR_NAME
              DOCKER_EXE             = $env:DOCKER_EXE
              AZ_EXE                 = $env:AZ_EXE
          }

          foreach ($item in $requiredVariables.GetEnumerator()) {
              if ([string]::IsNullOrWhiteSpace([string]$item.Value)) {
                  throw "Required variable '$($item.Key)' is not configured."
              }
          }

          if (-not (Test-Path -LiteralPath $env:DOCKER_EXE)) {
              throw "Docker executable not found: $env:DOCKER_EXE"
          }

          if (-not (Test-Path -LiteralPath $env:AZ_EXE)) {
              throw "Azure CLI executable not found: $env:AZ_EXE"
          }

          Write-Host "Storage account: $env:STORAGE_ACCOUNT_NAME"
          Write-Host "Storage container: $env:STORAGE_CONTAINER_NAME"
          Write-Host "Destination ACR: $env:ACR_NAME"
          Write-Host "Docker executable: $env:DOCKER_EXE"
          Write-Host "Azure CLI executable: $env:AZ_EXE"

      - name: Start and validate Docker service
        shell: powershell
        run: |
          $ErrorActionPreference = "Stop"

          $dockerService = Get-Service `
              -Name "docker" `
              -ErrorAction Stop

          if ($dockerService.Status -ne "Running") {
              Write-Host "Starting Docker service..."

              Start-Service -Name "docker"

              $dockerService.WaitForStatus(
                  "Running",
                  [TimeSpan]::FromSeconds(60)
              )
          }

          & "$env:DOCKER_EXE" version

          if ($LASTEXITCODE -ne 0) {
              throw "Docker Engine is not responding."
          }

      - name: Prepare working directories
        shell: powershell
        run: |
          $ErrorActionPreference = "Stop"

          $workDirectory = Join-Path `
              $env:RUNNER_TEMP `
              "container-import-$env:GITHUB_RUN_ID-$env:GITHUB_RUN_ATTEMPT"

          $imageTarPath = Join-Path `
              $workDirectory `
              "container-image.tar"

          $azureConfigDirectory = Join-Path `
              $workDirectory `
              "azure-cli"

          New-Item `
              -Path $workDirectory `
              -ItemType Directory `
              -Force | Out-Null

          New-Item `
              -Path $azureConfigDirectory `
              -ItemType Directory `
              -Force | Out-Null

          "WORK_DIRECTORY=$workDirectory" |
              Out-File `
                  -FilePath $env:GITHUB_ENV `
                  -Encoding utf8 `
                  -Append

          "IMAGE_TAR_PATH=$imageTarPath" |
              Out-File `
                  -FilePath $env:GITHUB_ENV `
                  -Encoding utf8 `
                  -Append

          # Keeps Azure CLI tokens isolated to this workflow run.
          "AZURE_CONFIG_DIR=$azureConfigDirectory" |
              Out-File `
                  -FilePath $env:GITHUB_ENV `
                  -Encoding utf8 `
                  -Append

          Write-Host "Working directory: $workDirectory"
          Write-Host "Local TAR path: $imageTarPath"

      - name: Azure login using service principal
        shell: powershell
        env:
          AZURE_CLIENT_ID: ${{ secrets.AZURE_CLIENT_ID }}
          AZURE_CLIENT_SECRET: ${{ secrets.AZURE_CLIENT_SECRET }}
          AZURE_TENANT_ID: ${{ secrets.AZURE_TENANT_ID }}
          AZURE_SUBSCRIPTION_ID: ${{ secrets.AZURE_SUBSCRIPTION_ID }}
        run: |
          $ErrorActionPreference = "Stop"

          $requiredSecrets = @{
              AZURE_CLIENT_ID       = $env:AZURE_CLIENT_ID
              AZURE_CLIENT_SECRET   = $env:AZURE_CLIENT_SECRET
              AZURE_TENANT_ID       = $env:AZURE_TENANT_ID
              AZURE_SUBSCRIPTION_ID = $env:AZURE_SUBSCRIPTION_ID
          }

          foreach ($item in $requiredSecrets.GetEnumerator()) {
              if ([string]::IsNullOrWhiteSpace([string]$item.Value)) {
                  throw "GitHub secret '$($item.Key)' is not configured."
              }
          }

          & "$env:AZ_EXE" login `
              --service-principal `
              --username "$env:AZURE_CLIENT_ID" `
              "--password=$env:AZURE_CLIENT_SECRET" `
              --tenant "$env:AZURE_TENANT_ID" `
              --output none `
              --only-show-errors

          if ($LASTEXITCODE -ne 0) {
              throw "Azure service principal login failed."
          }

          & "$env:AZ_EXE" account set `
              --subscription "$env:AZURE_SUBSCRIPTION_ID"

          if ($LASTEXITCODE -ne 0) {
              throw "Failed to select Azure subscription."
          }

          $account = & "$env:AZ_EXE" account show `
              --query "{subscription:name, subscriptionId:id, tenantId:tenantId}" `
              --output json `
              --only-show-errors

          if ($LASTEXITCODE -ne 0) {
              throw "Unable to validate the Azure login."
          }

          $account | Write-Host

      - name: Validate blob exists
        shell: powershell
        run: |
          $ErrorActionPreference = "Stop"

          $blobExists = & "$env:AZ_EXE" storage blob exists `
              --account-name "$env:STORAGE_ACCOUNT_NAME" `
              --container-name "$env:STORAGE_CONTAINER_NAME" `
              --name "${{ inputs.blob_name }}" `
              --auth-mode login `
              --query "exists" `
              --output tsv `
              --only-show-errors

          if ($LASTEXITCODE -ne 0) {
              throw "Failed to check the blob in Azure Storage."
          }

          if ($blobExists -ne "true") {
              throw @"
          Blob was not found.

          Storage account: $env:STORAGE_ACCOUNT_NAME
          Container:       $env:STORAGE_CONTAINER_NAME
          Blob:            ${{ inputs.blob_name }}
          "@
          }

          Write-Host "Blob found: ${{ inputs.blob_name }}"

      - name: Download image TAR from Storage
        shell: powershell
        run: |
          $ErrorActionPreference = "Stop"

          & "$env:AZ_EXE" storage blob download `
              --account-name "$env:STORAGE_ACCOUNT_NAME" `
              --container-name "$env:STORAGE_CONTAINER_NAME" `
              --name "${{ inputs.blob_name }}" `
              --file "$env:IMAGE_TAR_PATH" `
              --auth-mode login `
              --overwrite true `
              --only-show-errors

          if ($LASTEXITCODE -ne 0) {
              throw "Failed to download the TAR from Azure Storage."
          }

          if (-not (Test-Path -LiteralPath $env:IMAGE_TAR_PATH)) {
              throw "Downloaded TAR was not found: $env:IMAGE_TAR_PATH"
          }

          $file = Get-Item -LiteralPath $env:IMAGE_TAR_PATH

          if ($file.Length -eq 0) {
              throw "Downloaded TAR file is empty."
          }

          $hash = Get-FileHash `
              -LiteralPath $env:IMAGE_TAR_PATH `
              -Algorithm SHA256

          Write-Host "Downloaded size: $($file.Length) bytes"
          Write-Host "TAR SHA256: $($hash.Hash)"

      - name: Load TAR and detect image ID
        shell: powershell
        run: |
          $ErrorActionPreference = "Stop"

          $beforeImageIds = @(
              & "$env:DOCKER_EXE" image ls `
                  --all `
                  --no-trunc `
                  --quiet |
                  Sort-Object -Unique
          )

          Write-Host "Loading Docker image TAR..."

          $loadOutput = @(
              & "$env:DOCKER_EXE" load `
                  --input "$env:IMAGE_TAR_PATH" 2>&1
          )

          $loadExitCode = $LASTEXITCODE

          foreach ($line in $loadOutput) {
              Write-Host ([string]$line)
          }

          if ($loadExitCode -ne 0) {
              throw @"
          Docker failed to load the TAR.

          Confirm that it is a Docker image archive created using
          'docker save', not a container filesystem created using
          'docker export'.
          "@
          }

          $detectedImageIds = @()

          foreach ($line in $loadOutput) {
              $text = [string]$line

              # Example: Loaded image ID: sha256:abc123
              if ($text -match "Loaded image ID:\s*(sha256:[a-fA-F0-9]+)") {
                  $detectedImageIds += $Matches[1]
                  continue
              }

              # Also supports a TAR that unexpectedly contains a tag.
              if ($text -match "Loaded image:\s*(.+)$") {
                  $loadedReference = $Matches[1].Trim()

                  $inspectedId = & "$env:DOCKER_EXE" image inspect `
                      --format "{{.Id}}" `
                      "$loadedReference"

                  if ($LASTEXITCODE -eq 0 -and $inspectedId) {
                      $detectedImageIds += $inspectedId.Trim()
                  }
              }
          }

          $detectedImageIds = @(
              $detectedImageIds |
                  Where-Object { $_ } |
                  Sort-Object -Unique
          )

          # Fallback: compare Docker images before and after loading.
          if ($detectedImageIds.Count -eq 0) {
              $afterImageIds = @(
                  & "$env:DOCKER_EXE" image ls `
                      --all `
                      --no-trunc `
                      --quiet |
                      Sort-Object -Unique
              )

              $detectedImageIds = @(
                  $afterImageIds |
                      Where-Object { $_ -notin $beforeImageIds }
              )
          }

          if ($detectedImageIds.Count -eq 0) {
              throw "Docker loaded the TAR, but the image ID could not be detected."
          }

          if ($detectedImageIds.Count -gt 1) {
              Write-Host "Detected image IDs:"

              foreach ($imageId in $detectedImageIds) {
                  Write-Host " - $imageId"
              }

              throw "The TAR contains more than one image. One image per TAR is required."
          }

          $sourceImageId = $detectedImageIds[0]

          $sourceImagePreexisted = if (
              $sourceImageId -in $beforeImageIds
          ) {
              "true"
          }
          else {
              "false"
          }

          & "$env:DOCKER_EXE" image inspect "$sourceImageId"

          if ($LASTEXITCODE -ne 0) {
              throw "Detected image ID could not be inspected: $sourceImageId"
          }

          "SOURCE_IMAGE_ID=$sourceImageId" |
              Out-File `
                  -FilePath $env:GITHUB_ENV `
                  -Encoding utf8 `
                  -Append

          "SOURCE_IMAGE_PREEXISTED=$sourceImagePreexisted" |
              Out-File `
                  -FilePath $env:GITHUB_ENV `
                  -Encoding utf8 `
                  -Append

          Write-Host "Detected source image ID: $sourceImageId"

      - name: Obtain ACR access token and Docker login
        shell: powershell
        run: |
          $ErrorActionPreference = "Stop"

          $acrLoginJson = & "$env:AZ_EXE" acr login `
              --name "$env:ACR_NAME" `
              --expose-token `
              --output json `
              --only-show-errors

          if ($LASTEXITCODE -ne 0) {
              throw "Failed to obtain an ACR access token."
          }

          $acrLogin = $acrLoginJson | ConvertFrom-Json

          if ([string]::IsNullOrWhiteSpace($acrLogin.accessToken)) {
              throw "ACR login did not return an access token."
          }

          if ([string]::IsNullOrWhiteSpace($acrLogin.loginServer)) {
              throw "ACR login did not return a login server."
          }

          $acrLogin.accessToken |
              & "$env:DOCKER_EXE" login `
                  "$($acrLogin.loginServer)" `
                  --username "00000000-0000-0000-0000-000000000000" `
                  --password-stdin

          if ($LASTEXITCODE -ne 0) {
              throw "Docker login to ACR failed."
          }

          $targetImage = @(
              "$($acrLogin.loginServer)/"
              "${{ inputs.acr_repository }}"
              ":"
              "${{ inputs.image_tag }}"
          ) -join ""

          "ACR_LOGIN_SERVER=$($acrLogin.loginServer)" |
              Out-File `
                  -FilePath $env:GITHUB_ENV `
                  -Encoding utf8 `
                  -Append

          "TARGET_IMAGE=$targetImage" |
              Out-File `
                  -FilePath $env:GITHUB_ENV `
                  -Encoding utf8 `
                  -Append

          Write-Host "Destination image: $targetImage"

      - name: Tag image ID for ACR
        shell: powershell
        run: |
          $ErrorActionPreference = "Stop"

          if ([string]::IsNullOrWhiteSpace($env:SOURCE_IMAGE_ID)) {
              throw "SOURCE_IMAGE_ID was not detected."
          }

          & "$env:DOCKER_EXE" tag `
              "$env:SOURCE_IMAGE_ID" `
              "$env:TARGET_IMAGE"

          if ($LASTEXITCODE -ne 0) {
              throw "Failed to tag the image for ACR."
          }

          & "$env:DOCKER_EXE" image inspect "$env:TARGET_IMAGE"

          if ($LASTEXITCODE -ne 0) {
              throw "Destination ACR tag was not created."
          }

      - name: Push image to ACR
        shell: powershell
        run: |
          $ErrorActionPreference = "Stop"

          Write-Host "Pushing image: $env:TARGET_IMAGE"

          & "$env:DOCKER_EXE" push "$env:TARGET_IMAGE"

          if ($LASTEXITCODE -ne 0) {
              throw "Docker push to ACR failed."
          }

      - name: Verify image in ACR
        shell: powershell
        run: |
          $ErrorActionPreference = "Stop"

          $acrImage = @(
              "${{ inputs.acr_repository }}"
              ":"
              "${{ inputs.image_tag }}"
          ) -join ""

          & "$env:AZ_EXE" acr repository show `
              --name "$env:ACR_NAME" `
              --image "$acrImage" `
              --output table `
              --only-show-errors

          if ($LASTEXITCODE -ne 0) {
              throw "Image push completed, but ACR verification failed."
          }

          Write-Host "Image successfully pushed:"
          Write-Host "$env:TARGET_IMAGE"

      - name: Cleanup runner
        if: always()
        shell: powershell
        run: |
          if (
              $env:ACR_LOGIN_SERVER -and
              (Test-Path -LiteralPath $env:DOCKER_EXE)
          ) {
              & "$env:DOCKER_EXE" logout `
                  "$env:ACR_LOGIN_SERVER" 2>$null
          }

          if (
              $env:TARGET_IMAGE -and
              (Test-Path -LiteralPath $env:DOCKER_EXE)
          ) {
              & "$env:DOCKER_EXE" image rm `
                  --force `
                  "$env:TARGET_IMAGE" 2>$null
          }

          if (
              $env:SOURCE_IMAGE_ID -and
              $env:SOURCE_IMAGE_PREEXISTED -eq "false" -and
              (Test-Path -LiteralPath $env:DOCKER_EXE)
          ) {
              & "$env:DOCKER_EXE" image rm `
                  --force `
                  "$env:SOURCE_IMAGE_ID" 2>$null
          }

          if (Test-Path -LiteralPath $env:AZ_EXE) {
              & "$env:AZ_EXE" logout 2>$null
              & "$env:AZ_EXE" account clear 2>$null
          }

          if (
              $env:WORK_DIRECTORY -and
              (Test-Path -LiteralPath $env:WORK_DIRECTORY)
          ) {
              Remove-Item `
                  -LiteralPath $env:WORK_DIRECTORY `
                  -Recurse `
                  -Force `
                  -ErrorAction SilentlyContinue
          }

          Write-Host "Runner cleanup completed."

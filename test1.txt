name: Terraform Deploy

on:
  pull_request:
    branches:
      - main
    paths:
      - "terraform/**"

  push:
    branches:
      - main
    paths:
      - "terraform/**"

  workflow_dispatch:
    inputs:
      environment:
        description: "Target environment"
        required: true
        default: "dev"
        type: choice
        options:
          - dev
          - sit
          - ppd
          - prod

      action:
        description: "Terraform action"
        required: true
        default: "plan"
        type: choice
        options:
          - plan
          - apply
          - destroy-plan
          - destroy-apply

permissions:
  id-token: write
  contents: read
  pull-requests: write

env:
  TF_VERSION: "1.9.8"
  TF_WORKING_DIR: "terraform"
  ARM_USE_OIDC: true
  ARM_CLIENT_ID: ${{ vars.AZURE_CLIENT_ID }}
  ARM_TENANT_ID: ${{ vars.AZURE_TENANT_ID }}
  ARM_SUBSCRIPTION_ID: ${{ vars.AZURE_SUBSCRIPTION_ID }}

jobs:
  terraform-plan:
    name: Terraform Plan
    runs-on: ubuntu-latest

    environment: ${{ github.event.inputs.environment || 'dev' }}

    defaults:
      run:
        working-directory: ${{ env.TF_WORKING_DIR }}

    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Setup Terraform
        uses: hashicorp/setup-terraform@v3
        with:
          terraform_version: ${{ env.TF_VERSION }}

      - name: Terraform Format Check
        run: terraform fmt -check -recursive

      - name: Terraform Init
        run: |
          terraform init \
            -backend-config=backend/${{ github.event.inputs.environment || 'dev' }}.hcl

      - name: Terraform Validate
        run: terraform validate

      - name: Terraform Plan
        if: ${{ github.event.inputs.action != 'destroy-plan' && github.event.inputs.action != 'destroy-apply' }}
        run: |
          terraform plan \
            -var-file=env/${{ github.event.inputs.environment || 'dev' }}.tfvars \
            -out=tfplan

      - name: Terraform Destroy Plan
        if: ${{ github.event.inputs.action == 'destroy-plan' || github.event.inputs.action == 'destroy-apply' }}
        run: |
          terraform plan \
            -destroy \
            -var-file=env/${{ github.event.inputs.environment || 'dev' }}.tfvars \
            -out=tfplan

      - name: Upload Terraform Plan
        uses: actions/upload-artifact@v4
        with:
          name: tfplan-${{ github.event.inputs.environment || 'dev' }}
          path: ${{ env.TF_WORKING_DIR }}/tfplan
          retention-days: 1

  terraform-apply:
    name: Terraform Apply
    runs-on: ubuntu-latest
    needs: terraform-plan

    if: |
      github.event_name == 'workflow_dispatch' &&
      (
        github.event.inputs.action == 'apply' ||
        github.event.inputs.action == 'destroy-apply'
      )

    environment: ${{ github.event.inputs.environment }}

    defaults:
      run:
        working-directory: ${{ env.TF_WORKING_DIR }}

    steps:
      - name: Checkout repo
        uses: actions/checkout@v4

      - name: Setup Terraform
        uses: hashicorp/setup-terraform@v3
        with:
          terraform_version: ${{ env.TF_VERSION }}

      - name: Download Terraform Plan
        uses: actions/download-artifact@v4
        with:
          name: tfplan-${{ github.event.inputs.environment }}
          path: ${{ env.TF_WORKING_DIR }}

      - name: Terraform Init
        run: |
          terraform init \
            -backend-config=backend/${{ github.event.inputs.environment }}.hcl

      - name: Terraform Apply
        run: terraform apply -auto-approve tfplan

        name: Import Image from Storage to ACR

on:
  workflow_dispatch:
    inputs:
      storage_account:
        description: "Storage account name"
        required: true
        default: "mystorageacct"

      storage_container:
        description: "Storage container name"
        required: true
        default: "hcp-images"

      blob_name:
        description: "Image tar blob name"
        required: true
        default: "myapp.tar"

      local_image_name:
        description: "Image name after docker load"
        required: true
        default: "myapp:latest"

      acr_name:
        description: "Target ACR name"
        required: true
        default: "companyacr"

      acr_image:
        description: "Target ACR image path"
        required: true
        default: "apps/myapp:1.0.0"

permissions:
  id-token: write
  contents: read

jobs:
  import-image-to-acr:
    runs-on:
      - self-hosted
      - windows

    steps:
      - name: Azure Login
        uses: azure/login@v2
        with:
          client-id: ${{ secrets.AZURE_CLIENT_ID }}
          tenant-id: ${{ secrets.AZURE_TENANT_ID }}
          subscription-id: ${{ secrets.AZURE_SUBSCRIPTION_ID }}

      - name: Prepare local folder
        shell: powershell
        run: |
          New-Item -ItemType Directory -Force -Path "C:\Temp\container-images"

      - name: Download image tar from Storage Account
        shell: powershell
        run: |
          az storage blob download `
            --account-name "${{ github.event.inputs.storage_account }}" `
            --container-name "${{ github.event.inputs.storage_container }}" `
            --name "${{ github.event.inputs.blob_name }}" `
            --file "C:\Temp\container-images\image.tar" `
            --auth-mode login `
            --overwrite true

      - name: Load Docker image
        shell: powershell
        run: |
          docker load -i "C:\Temp\container-images\image.tar"

      - name: Validate loaded image
        shell: powershell
        run: |
          docker image inspect "${{ github.event.inputs.local_image_name }}"

      - name: Login to ACR
        shell: powershell
        run: |
          az acr login --name "${{ github.event.inputs.acr_name }}"

      - name: Tag image for ACR
        shell: powershell
        run: |
          docker tag `
            "${{ github.event.inputs.local_image_name }}" `
            "${{ github.event.inputs.acr_name }}.azurecr.io/${{ github.event.inputs.acr_image }}"

      - name: Push image to ACR
        shell: powershell
        run: |
          docker push `
            "${{ github.event.inputs.acr_name }}.azurecr.io/${{ github.event.inputs.acr_image }}"

      - name: Verify image in ACR
        shell: powershell
        run: |
          $repository = "${{ github.event.inputs.acr_image }}".Split(":")[0]

          az acr repository show-tags `
            --name "${{ github.event.inputs.acr_name }}" `
            --repository $repository `
            --output table

      - name: Cleanup
        if: always()
        shell: powershell
        run: |
          Remove-Item -Path "C:\Temp\container-images" -Recurse -Force -ErrorAction SilentlyContinue
